 *   `empty`    no preset is filled yet;  `filled`  every slot is filled
 *   `edit`     a preset is being built (slot 1)
 *   `sent`     preset 1 was just sent (with `room`: part of it fitted)
 *   `room`     the queue has two free stacks (it is full without)
 *   `remote`   an outpost run from home (level 3);  `yards`  the yard page
 *   `housing`  the tab's own housing box is open
 *
 * - Presets are numbered, never named: 3 slots at level 2, 5 at level 3.
 * - A preset is made only here: tap a slot, pick monsters and counts.
 * - Send adds the preset to the END of the queue; when the queue or the goo
 *   runs out it adds what fits and says how many ("Added 82 of 140").
 * - Level 3 keeps two lists: the main yard's, and one shared by every outpost.
 */

const QUEUE_STACKS = 10;

type Preset = [id: string, count: number][];
type Scope = "main" | "outposts";

const MAIN_PRESETS: readonly Preset[] = [
  [["C8", 60], ["C7", 40], ["C1", 40]],

];
/** What a first preset looks like half-way through being built. */
const BUILDING: Preset = [["C8", 40], ["C7", 20]];

const busyQueue = (size: number, stacks: number): QueueStack[] =>
  (
    [
      ["C12", size - 1],
      ["C12", size],
      ["C8", size],
      ["C8", size],
      ["C7", size],
      ["C14", size],
      ["C5", size],
      ["C3", size],
      ["C1", size],
      ["C6", Math.round(size / 2)],
    ] as const
  )
    .slice(0, stacks)
    .map(([id, count]) => [id, count, 6] as QueueStack);

const outpostQueue = (size: number): QueueStack[] =>
  (
    [
      ["C7", size],
      ["C7", size],
      ["C2", size],
      ["C2", size],
      ["C1", size - 4],
    ] as const
  ).map(([id, count]) => [id, count, 4] as QueueStack);

const el = <K extends keyof HTMLElementTagNameMap>(tag: K, className: string, text?: string): HTMLElementTagNameMap[K] => {
  const node = document.createElement(tag);
  node.className = className;
  if (text !== undefined) node.textContent = text;
  return node;
};

const btn = (className: string, label: string, ...content: (Node | string)[]): HTMLButtonElement => {
  const node = el("button", className);
  node.type = "button";
  node.setAttribute("aria-label", label);
  node.append(...content);
  return node;
};

const portrait = (id: string, className: string): Node => {
  const entry = monsterEntry(id);
  return entry ? monsterPicture(entry, "small", className) : el("span", className);
};

const art = (name: GameIconName, size: number, className = ""): HTMLImageElement =>
  gameIcon(name, { size: size as 16 | 24 | 32, decorative: true, ...(className ? { className } : {}) });

const arrow = (direction: -1 | 1): HTMLImageElement =>
  art(`friend${direction === 1 ? "next" : "prev"}${classicLook() ? "classic" : ""}` as GameIconName, 32, "b9r2h-arrow");

const short = (value: number): string =>
  value >= 1_000_000 ? `${(value / 1_000_000).toFixed(1)}M` : value >= 1_000 ? `${Math.round(value / 1000)}K` : value.toLocaleString("en-US");

const totalOf = (preset: Preset): number => preset.reduce((sum, [, count]) => sum + count, 0);

  /** The yard the tab shows: the fake busiest queue, or the outpost being run from home. */
  fakeYard(yard: HatchYard): HatchYard {
    const scope = this.scope();
    if (this.remote) {
      const endsAt = Math.floor(Date.now() / 1000) + 620;
      const hatcheries = yard.hatcheries
        .slice(0, 2)

      return { ...yard, hatcheries, shared: this.queues[scope], capacity: 9_600, housed: { C7: 120, C2: 80 }, goo: 18_400_000 - this.spent[scope] };

    return { ...yard, shared: this.queues[scope], goo: yard.goo - this.spent[scope] };

  private yards(): YardRow[] {

  /** Adds the preset to the end of the queue by the HCC's stacking rule, as far as stacks and goo go. */
  private send(at: number): void {
    const scope = this.scope();
    const preset = this.presets[scope][at];
    const yard = this.yard;
    if (!preset?.length || !yard) return;
    const queue = this.queues[scope];
    let goo = yard.goo;
    let added = 0;
    let why: Sent["why"] = null;
    outer: for (const [id, count] of preset) {
      const price = this.priceOf(id);
      const level = academyLevels(this.save())[id] ?? 1;
      for (let n = 0; n < count; n += 1) {
        if (goo < price) {
          why = "goo";
          break outer;
        }
        const last = queue.at(-1);
        if (last && last[0] === id && last[1] < this.size) last[1] += 1;
        else if (queue.length < QUEUE_STACKS) queue.push([id, 1, level]);
        else {
          why = "queue";
          break outer;
        }
        goo -= price;
        added += 1;
      }
    }
    this.spent[scope] += yard.goo - goo;
    this.sent = { scope, slot: at, added, of: totalOf(preset), why };
    this.redraw();
  }

  /* ── Building a preset: the monsters and the count, for slot N ────── */

  private edit(at: number): void {
    this.editing = at;
    this.sent = null;
    this.renderTabs();
    this.renderEditor();
    // The count shows the chosen monster's share of the preset.
    this.scroll.querySelector<HTMLElement>(".hatch-monster[aria-pressed=true]")?.scrollIntoView({ block: "nearest", inline: "nearest" });
  }

  private chosen(): string | null {
    return this.root.querySelector<HTMLElement>(".hatch-monster[aria-pressed=true]")?.dataset.monster ?? null;
  }

  private setCount(id: string, count: number): void {
    if (this.editing === null) return;
    const preset = this.presets[this.scope()][this.editing]!;
    const other = preset.filter(([one]) => one !== id);
    const room = (QUEUE_STACKS - this.stacksOf(other)) * this.size;
    const next = Math.max(0, Math.min(Math.floor(count) || 0, room));
    const at = preset.findIndex(([one]) => one === id);
    if (next === 0) {
      if (at >= 0) preset.splice(at, 1);
    } else if (at >= 0) preset[at]![1] = next;
    else preset.push([id, next]);
    this.renderEditor();
  }

  private renderEditor(): void {
    const at = this.editing;
    this.editHead.hidden = at === null;
    this.editBox.hidden = at === null;
    for (const mark of this.root.querySelectorAll(".b9r3h-inpreset")) mark.remove();
    if (at === null) return;
    const scope = this.scope();
    const preset = this.presets[scope][at]!;
    const chosen = this.chosen();

    // The head: the slot's number (and whose list), what is in it, its size, Clear, Done.

  /**
   * Runs a redraw with the scroll box held where the player left it (owner,
   * 9 and 13 Nov: on a phone, choosing a monster, `+` or `+5` threw the list
   * back to the top). Every redraw comes through here, the whole one and the
   * count's alone. Only a hatchery tab's tap opens at the top, and it sets
   * that before it redraws.
   *
   * - The focus: a redraw that took away the node with the focus gives it to
   *   the new copy without a scroll. A plain focus() scrolls the box to it,
   *   and on a phone that is often up in the line.
   * - The box's height: a shorter info panel (D.A.V.E.'s after Crabatron's)
   *   made the box shorter while it was scrolled to the bottom, and the
   *   browser pulled the scroll up (tester, 9 Nov). The panel keeps the
   *   height it lost.
   * - Late moves: the browser lays the new nodes out, and decodes their
   *   pictures, after this returns. For the next frames, a box that moved
   *   without the player scrolling it goes back.
   */
  private keepScroll(draw: () => void): void {
    const box = this.scroll;
    const focused = document.activeElement as HTMLElement | null;
    const focusKey = focused?.dataset?.["focusKey"];
    const scrolled = box.scrollTop;
    const tall = box.scrollHeight;
    // The info panel's held height is let go; it is measured again.
    this.info.style.minHeight = "";
    draw();
    if (focusKey && focused && !focused.isConnected) {
      this.element.querySelector<HTMLElement>(`[data-focus-key="${focusKey}"]`)?.focus({ preventScroll: true });
    }
    this.holdScroll(scrolled, tall);
    this.settleScroll(scrolled, tall);
  }

  /** Puts the box back at `scrolled`, the info panel keeping any height the box lost. */
  private holdScroll(scrolled: number, tall: number): void {
    const box = this.scroll;
    if (box.scrollTop === scrolled) return;
    const lost = tall - box.scrollHeight;
    if (lost > 0) this.info.style.minHeight = `${this.info.offsetHeight + lost}px`;
    box.scrollTop = scrolled;
  }

  /** For the next frames, a box that moved on its own goes back (see `keepScroll`). */
  private settleScroll(scrolled: number, tall: number): void {
    if (typeof requestAnimationFrame !== "function") return;
    if (this.settling !== null) cancelAnimationFrame(this.settling);
    this.settling = null;
    const since = this.scrolledAt;
    // A list still gliding from a fling moves by itself: leave it be.
    if (performance.now() - since < GLIDE_MS) return;
    let frames = SETTLE_FRAMES;
    const check = (): void => {
      this.settling = null;
      // The player scrolled meanwhile: where they put it is where it stays.
      if (!this.scroll.isConnected || this.scrolledAt !== since) return;
      this.holdScroll(scrolled, tall);
      frames -= 1;
      if (frames > 0) this.settling = requestAnimationFrame(check);
    };
    this.settling = requestAnimationFrame(check);
  }

  private renderGrid(monsters: readonly HatchMonster[]): void {

    if (order !== this.gridOrder) {
      this.gridOrder = order;
      const ids = new Set(monsters.map((row) => row.monster.id));
      for (const id of [...this.tiles.keys()]) if (!ids.has(id)) this.tiles.delete(id);
      this.grid.replaceChildren(...monsters.map((row) => this.tile(row).item));
    }
    for (const row of monsters) {
      const tile = this.tile(row);
      const { button, note } = tile;
      const kind = `hatch-monster hatch-monster--${row.state.kind}`;
      if (button.className !== kind) button.className = kind;
      button.setAttribute("aria-pressed", String(row.monster.id === this.selected));
      // The tile says its price short ("468K", owner 9 Nov: easier to see), rounded up so it never looks cheaper;

      const noteKey = row.state.kind === "ready" ? `ready ${row.price}` : row.state.kind;

    const item = document.createElement("li");
    const button = document.createElement("button");
    button.type = "button";
    button.dataset["monster"] = id;
    button.dataset["focusKey"] = `monster-${id}`;
    button.addEventListener("click", () => this.tapMonster(id));
    const name = document.createElement("span");
    // A long name ("Slimeattikus") gets smaller letters on a phone, so it
    // stays whole on one line rather than breaking mid-word.
    // Side by side, nine across, a name of nine letters or more is drawn a
    // little smaller too, so none is cut with "…" (patch 7.6.1, astra).
    const letters = row.monster.name.length;
    name.className = `hatch-monster__name${letters > LONG_NAME ? " hatch-monster__name--long" : ""}${letters > MID_NAME ? " hatch-monster__name--mid" : ""}`;
    name.textContent = row.monster.name;
    const note = document.createElement("span");
    note.className = "hatch-monster__note";
    button.append(monsterPicture(row.monster, "small", "hatch-monster__picture"), name, " ", note);
    item.append(button);
    const tile: HatchTile = { item, button, note, noteKey: "" };
    this.tiles.set(id, tile);
    return tile;
  }

export const gameIconSrcset = (name: GameIconName): string =>

  readonly label?: string;

  readonly className?: string;

  if (options.decorative) {
    img.alt = "";
    img.setAttribute("aria-hidden", "true");
  } else {
    const word = options.label ?? GAME_ICONS[name];
    img.alt = word;
    img.setAttribute("role", "img");
    img.setAttribute("aria-label", word);
    if (options.tooltip !== false) img.title = word;
  }
  return img;
};

export interface IconAmountOptions extends GameIconOptions {
  /** Spell a number short ("15.0M") instead of in full ("15,000,000"). */
  readonly compact?: boolean;
  /** Draw it as a `.chip`: the rounded label of a level or a count. */
  readonly chip?: boolean;
  /**
   * Something that happened, not a price or a level: "+300" in green for a
   * gain, "−300" in red for a loss, said as "300 gained" / "300 lost"
   * (`amountChange.ts`).
   */
  readonly change?: AmountChange;
}

/**
 * "icon 480": a painted icon and a number, kept together on one line; the
 * `resourceAmount` (resourceIcon.ts) of everything that is not a resource.
 * The icon is 16 px unless `size` says otherwise.
 *
 * A number is spelled in full unless `compact` asks otherwise. A string is
 * taken as already spelled, for "400 → 480", "35%" or "0/3". `change` signs
 * and colours it as a gain or a loss.
 */
export const iconAmount = (
  name: GameIconName,
  amount: number | string,
  options: IconAmountOptions = {},
): HTMLElement => {
  const wrapper = document.createElement("span");
  wrapper.className = ["icon-amount", options.chip ? "chip" : "", options.className ?? ""].filter(Boolean).join(" ");
  wrapper.dataset["icon"] = name;
  const icon = gameIcon(name, {
    size: options.size ?? 16,
    ...(options.label === undefined ? {} : { label: options.label }),
    ...(options.decorative === undefined ? {} : { decorative: options.decorative }),
    ...(options.tooltip === undefined ? {} : { tooltip: options.tooltip }),
  });
  wrapper.append(icon);
  writeAmount(wrapper, "icon-amount", amount, options);
  return wrapper;
};

/** {@link iconAmount} as a chip: "[heart 480]". */
export const iconChip = (
  name: GameIconName,
  amount: number | string,
  options: Omit<IconAmountOptions, "chip"> = {},
): HTMLElement => iconAmount(name, amount, { ...options, chip: true });
